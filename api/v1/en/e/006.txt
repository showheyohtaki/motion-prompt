// Typewriter / タイプライター
// Pack: motion (Motion graphics moves) · element 06 · Category: Entrances & text
// What happens: Text is typed into an input field one character at a time, then the send button lights up and is pressed.
// Motion spec:
//   - One pass per 4.7 s. Nine characters appear in a white input field (471 × 92, radius 30), one every 0.24 s starting at 0.3 s (typing ends at 2.46 s).
//   - Each new character settles from scale 1.35 to 1: 1 + 0.35·e^(−16t).
//   - Cursor (coral, 5 × 44) sits after the last character: solid while typing, otherwise blinking with a 0.94 s period (on for 0.5 s).
//   - Send button (radius 30): grey, then a blue disc appears from 2.58 s with spring 1 − e^(−6.5t)·cos(15t).
//   - Press: the button shrinks by up to 20% at 2.96–3.21 s (half sine). At 3.13–3.41 s the text moves 70 px up while fading out.
//   - The cursor returns to the start once the text is half faded (about 3.27 s); at 3.51–3.81 s the blue fades.
// When to use it: Use it to show someone typing: a prompt, a search query, a chat message.
// Tags: typewriter, typing, text, cursor, input, prompt, chat, type-on
// Size: 539x208 px · Cycle: 4.7 s
// Needs: kit motion -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/motion.js
// Live demo: https://showheyohtaki.github.io/motion-prompt/#006
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 06
// 06 タイプライター（入力欄に1文字ずつ打たれ、打ち終わると送信ボタンが点く）
CELLS[5] = (k, w, h, u) => {
  stage(k);
  const S = mpText('ねこの動画つくって'), N = [...S].length, t0 = 0.3, dt = 0.24, done = t0 + N * dt;
  const BX = 34, BY = 58, BW = w - 68, BH = 92;
  const sendT = done + 0.55, clr = P(u, sendT + 0.12, sendT + 0.4);
  k.fillStyle = '#fff'; rr(k, BX, BY, BW, BH, 30); k.fill();
  const f = `600 36px ${JP}`; k.font = f;
  let n = clamp(Math.floor((u - t0) / dt) + 1, 0, N); if (u < t0) n = 0;
  let x = BX + 32; const y = BY + BH / 2 + 2;
  k.save(); rr(k, BX, BY, BW - 86, BH, 30); k.clip();
  [...S].slice(0, n).forEach((ch, i) => {
    const age = u - (t0 + i * dt), pop = 1 + 0.35 * Math.exp(-age * 16), cw = k.measureText(ch).width;
    k.save(); k.translate(x + cw / 2, y - clr * 70); k.scale(pop, pop); k.globalAlpha = 1 - clr;
    txt(k, ch, 0, 0, f, C.navy, 'center', 'middle'); k.restore(); k.font = f; x += cw + 1;
  });
  k.restore();
  if (clr > 0.5) x = BX + 32;
  // カーソル（打っている間は点灯。待ちの間は0.94秒周期でゆっくり明滅）
  const typing = u >= t0 - 0.05 && u < done + 0.1, blink = typing ? 1 : (mod(u, 0.94) < 0.5 ? 1 : 0);
  if (blink) { k.fillStyle = C.coral; rr(k, x + 4, y - 23, 5, 44, 2.5); k.fill(); }
  // 送信ボタン
  const on = spring(u - done - 0.12, 15, 6.5) * (1 - P(u, sendT + 0.5, sendT + 0.8)), press = 1 - 0.2 * Math.sin(Math.PI * P(u, sendT - 0.05, sendT + 0.2));
  const bx = BX + BW - 48, by = BY + BH / 2;
  k.fillStyle = '#DDE3F3'; k.beginPath(); k.arc(bx, by, 30, 0, 7); k.fill();
  if (on > 0.01) { k.fillStyle = C.blue; k.beginPath(); k.arc(bx, by, 30 * Math.min(on, 1.25) * press, 0, 7); k.fill(); }
  k.strokeStyle = '#fff'; k.lineWidth = 6; k.lineCap = 'round'; k.lineJoin = 'round'; k.beginPath(); k.moveTo(bx, by + 12); k.lineTo(bx, by - 12); k.moveTo(bx - 11, by - 2); k.lineTo(bx, by - 13); k.lineTo(bx + 11, by - 2); k.stroke();
};
