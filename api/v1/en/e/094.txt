// Marquee / マーキー
// Pack: lp (Landing page & web effects) · element 22 · Category: Loading, navigation & ambient loops
// What happens: Inside a navy band, large service names separated by orange dots keep moving to the left without a gap.
// Motion spec:
//   - Band: 58 tall, navy, full viewport width. Text: 30 px, heavy weight, 1 px letter spacing, white.
//   - One set is four words (WEB DESIGN / BRANDING / MOTION / UI/UX), 64 px apart, with an orange dot (radius 6) in each gap.
//   - Each cycle (4.8 s) moves left by the width of one set (about 881 px here) at constant speed. The set is measured and the offset wraps by that width, so there is no visible seam.
//   - The set is repeated until the viewport width is filled.
// When to use it: Use it to show service names, client logos or a tagline with energy.
// Tags: marquee, ticker, infinite-loop, scrolling-text, logo-strip, band, loop
// Size: 539x208 px · Cycle: 4.8 s
// Needs: kit lp -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp.js
// Live demo: https://showheyohtaki.github.io/motion-prompt/#094
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 22
// 22 マーキー（文字の帯が、途切れずに横へ流れ続ける）
CELLS[21] = (k, w, h, u) => {
  frame(k);
  const WORDS = ['WEB DESIGN', 'BRANDING', 'MOTION', 'UI/UX'], F = `800 30px ${EN}`, GAPW = 64;
  view(k, () => {
    bars(k, 34, 20, 180, 1, C.ink, 10); bars(k, 34, 38, 300, 1);
    k.fillStyle = C.ink; k.fillRect(0, 58, VW, 58);
    k.font = F; k.letterSpacing = '1px'; const ws = WORDS.map(s => k.measureText(mpText(s)).width); k.letterSpacing = '0px';
    const unit = ws.reduce((a, b) => a + b + GAPW, 0);
    let x = -mod(u / CYC, 1) * unit;                 // 1周期で1組ぶん左へ
    for (let rep = 0; x < VW; rep++) WORDS.forEach((s, i) => {
      txt(k, s, x, 88, F, '#fff', 'left', 'middle', '1px');
      k.fillStyle = C.acc; k.beginPath(); k.arc(x + ws[i] + GAPW / 2, 88, 6, 0, 7); k.fill();
      x += ws[i] + GAPW;
    });
    for (let i = 0; i < 3; i++) { k.fillStyle = C.soft; rr(k, 34 + i * 152, 130, 136, 30, 6); k.fill(); bars(k, 46 + i * 152, 142, 90, 1, C.faint); }
  });
};
