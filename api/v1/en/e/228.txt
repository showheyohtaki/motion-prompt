// Radar chart / レーダーチャート
// Pack: diagram (Diagrams and charts for slides) · element 12 · Category: Charts that show numbers
// What happens: On a six-axis spider web, a gray polygon spreads out from the center, then a blue one spreads with a slight overshoot.
// Motion spec:
//   - Scale: four nested hexagons up to radius 86 px (the outer one darker) and six spokes; they stay for the whole cycle
//   - Comparison: a gray polygon (values 0.46–0.74, fill opacity 0.22) spreads out from the center from 0.4 s to 1.1 s (ease-out)
//   - Main: a blue polygon (values 0.92, 0.58, 0.84, 0.4, 0.76, 0.52; fill opacity 0.28, 3 px outline, dots on the corners) spreads from 0.95 s to 1.85 s, a 55:45 blend of ease-out and ease-out-back for a slight overshoot
//   - End: the axes and grid stay; only the data fades out from 4.15 s to 4.55 s (one pass = 4.7 s)
// When to use it: To compare the balance of several items, or the strengths and weaknesses of two things.
// Tags: radar chart, spider chart, web chart, polygon, compare, skills, balance
// Size: 539x208 px · Cycle: 4.7 s
// Needs: kit diagram -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/diagram.js
// Live demo: https://showheyohtaki.github.io/motion-prompt/#228
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 12
// 12 レーダーチャート: a spider-web scale; a gray polygon, then the blue polygon spreads out from the center
const RA = [0.92, 0.58, 0.84, 0.4, 0.76, 0.52];
const RB = [0.55, 0.74, 0.46, 0.72, 0.5, 0.66];
function radarPath(k, cx, cy, R, vals, s) {
  k.beginPath();
  vals.forEach((v, i) => {
    const a = -Math.PI / 2 + i / vals.length * Math.PI * 2, r = R * v * s;
    const x = cx + Math.cos(a) * r, y = cy + Math.sin(a) * r;
    i ? k.lineTo(x, y) : k.moveTo(x, y);
  });
  k.closePath();
}
function radar(k, w, h, u) {
  bg(k, w, h);
  const cx = w / 2, cy = h / 2 + 2, R = 86, n = 6;
  k.lineJoin = 'round';
  for (let ring = 1; ring <= 4; ring++) {
    radarPath(k, cx, cy, R, Array(n).fill(ring / 4), 1);
    k.strokeStyle = ring === 4 ? C.g4 : C.g3; k.lineWidth = 2; k.stroke();
  }
  for (let i = 0; i < n; i++) {
    const a = -Math.PI / 2 + i / n * Math.PI * 2;
    seg(k, cx, cy, cx + Math.cos(a) * R, cy + Math.sin(a) * R, C.g3, 2);
    dot(k, cx + Math.cos(a) * R, cy + Math.sin(a) * R, 4, C.g4);
  }
  const fo = dataFade(u);
  const sb = easeOut(P(u, 0.4, 1.1));
  if (sb > 0.01) {
    k.save(); k.globalAlpha = fo;
    radarPath(k, cx, cy, R, RB, sb);
    k.fillStyle = 'rgba(169,168,165,0.22)'; k.fill();
    k.strokeStyle = C.g4; k.lineWidth = 2.5; k.stroke();
    k.restore();
  }
  const qa = P(u, 0.95, 1.85), sa = lerp(easeOut(qa), easeOutBack(qa), 0.45);   // mild overshoot, stays near the web
  if (sa > 0.01) {
    k.save(); k.globalAlpha = fo;
    radarPath(k, cx, cy, R, RA, sa);
    k.fillStyle = 'rgba(47,107,235,0.28)'; k.fill();
    k.strokeStyle = C.blue; k.lineWidth = 3; k.stroke();
    RA.forEach((v, i) => {
      const a = -Math.PI / 2 + i / n * Math.PI * 2, r = R * v * sa;
      dot(k, cx + Math.cos(a) * r, cy + Math.sin(a) * r, 4.5, C.blue);
    });
    k.restore();
  }
}
CELLS[11] = radar;
