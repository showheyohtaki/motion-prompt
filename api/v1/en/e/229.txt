// Gauge chart / ゲージチャート
// Pack: diagram (Diagrams and charts for slides) · element 13 · Category: Charts that show numbers
// What happens: The needle of a half-circle dial swings past, wobbles and settles at 72%; the blue band and the number follow the needle.
// Motion spec:
//   - Dial: a half-circle band of radius 120 px and 30 px thick (gray), with ticks every 10% (longer at 0, 50 and 100%)
//   - Needle: starts at 0.5 s and follows a damped spring 1 − e^(−3.2t)·cos(7.5t), swinging past and wobbling before it settles at 72%
//   - Band and number: the band is blue up to the needle, and the percentage (26 px) below matches the needle
//   - End: from 4.1 s to 4.5 s the needle, band and number go back to 0% (cubic ease-in-out; one pass = 4.7 s)
// When to use it: To show with one number how far you are toward a target.
// Tags: gauge, gauge chart, speedometer, dial, kpi, progress, needle
// Size: 539x208 px · Cycle: 4.7 s
// Needs: kit diagram -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/diagram.js
// Live demo: https://showheyohtaki.github.io/motion-prompt/#229
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 13
// 13 ゲージチャート: a half-circle meter; the needle swings past and settles at 72%
function gauge(k, w, h, u) {
  bg(k, w, h);
  const cx = w / 2, cy = 152, R = 120, T = 30, V = 0.72;
  const t = Math.max(0, u - 0.5);
  const s = u <= 0.5 ? 0 : 1 - Math.exp(-3.2 * t) * Math.cos(7.5 * t);   // overshoot, wobble, settle
  const f = V * s * (1 - easeInOut(P(u, 4.1, 4.5)));
  k.lineCap = 'butt';
  k.strokeStyle = C.g2; k.lineWidth = T;
  k.beginPath(); k.arc(cx, cy, R - T / 2, Math.PI, Math.PI * 2); k.stroke();
  if (f > 0.002) {
    k.strokeStyle = C.blue; k.lineWidth = T;
    k.beginPath(); k.arc(cx, cy, R - T / 2, Math.PI, Math.PI + Math.PI * f); k.stroke();
  }
  for (let i = 0; i <= 10; i++) {   // ticks every 10%
    const a = Math.PI + Math.PI * i / 10, r0 = R - T - 12, r1 = R - T - (i % 5 === 0 ? 24 : 18);
    seg(k, cx + Math.cos(a) * r0, cy + Math.sin(a) * r0, cx + Math.cos(a) * r1, cy + Math.sin(a) * r1, C.g4, 2.5);
  }
  const a = Math.PI + Math.PI * f, L = R - 22, bw = 7;
  const ca = Math.cos(a), sa = Math.sin(a);
  k.fillStyle = C.mark; k.beginPath();
  k.moveTo(cx + ca * L, cy + sa * L);
  k.lineTo(cx - sa * bw, cy + ca * bw);
  k.lineTo(cx - ca * 12, cy - sa * 12);
  k.lineTo(cx + sa * bw, cy - ca * bw);
  k.closePath(); k.fill();
  dot(k, cx, cy, 13, C.mark);
  dot(k, cx, cy, 5, C.stage);
  txt(k, `${Math.round(f * 100)}%`, cx, cy + 36, `700 26px ${FONT}`, C.mark, 'center', 'middle');
}
CELLS[12] = gauge;
