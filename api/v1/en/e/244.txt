// Multi-scroll (split scrolling) / マルチスクロール
// Pack: lp3 (Landing page & web effects, part 3) · element 04 · Category: Scroll & drag effects
// What happens: The screen is split into two columns. On scroll, the colored headings on the left move up and the pictures on the right move down, so three pairs line up in turn, then it runs back to the first pair.
// Motion spec:
//   - A 511 × 170 viewport split into two 255.5 px columns; each pair is one screen high (170). One pass takes 4.7 s (twice per 9.4 s pack cycle).
//   - Pair index s: 0→1 over 0.35–1.15 s, 1→2 over 1.95–2.75 s, 2→0 over 3.6–4.45 s (cubic ease-in-out).
//   - The left column places pair i at y = (i − s) × 170, the right column at y = (s − i) × 170 (its pairs are stacked bottom-up, so it moves the other way). Each change moves both columns by 170 px.
//   - Pairs: 01 COLOR (blue #2F6BEB), 02 SHAPE (light blue #C9D8FB), 03 MOTION (mid blue #9DB8F6). A white divider in the middle, three dots for the current pair, and opposite arrows while it moves.
//   - Not the same as #144 Split screen (showing parts side by side). Multi-scroll moves the split columns in opposite directions on scroll.
// When to use it: Use it for brand or project pages that pair one image with one heading per screen and want a striking change between them.
// Tags: multiscroll, split-scrolling, opposite, columns, scroll, fullpage
// Size: 539x208 px · Cycle: 9.4 s
// Needs: kit lp3 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp3.js
// Live demo: https://showheyohtaki.github.io/motion-prompt/#244
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 04
// 04 マルチスクロール（画面を左右2列に分け、1回のスクロールで左の列は上へ、右の列は下へ動いて、次の組がそろう）
CELLS[3] = (k, w, h, u) => {
  u = snapU(u);
  frame(k, 0);
  const s = steps([[0.35, 0], [1.15, 1], [1.95, 1], [2.75, 2], [3.6, 2], [4.45, 0]], u);   // いまの組（0〜2）
  const HW = VW / 2;
  const SEC = [[C.blue, '01', 'COLOR', 0], [C.blueL, '02', 'SHAPE', 1], [C.blueM, '03', 'MOTION', 3]];
  view(k, () => {
    // 左の列: 0,1,2 の順に上から並び、上へ流れる
    k.save(); k.beginPath(); k.rect(0, 0, HW, VH); k.clip();
    SEC.forEach(([col, no, word], i) => {
      const y = (i - s) * VH;
      if (y > VH || y < -VH) return;
      k.fillStyle = col; k.fillRect(0, y, HW, VH);
      const dark = col === C.blueL || col === C.blueM;
      txt(k, no, 28, y + 44, `700 12px ${MONO}`, dark ? C.ink : 'rgba(255,255,255,.75)');
      txt(k, word, 26, y + 96, `900 40px ${EN}`, dark ? C.ink : '#fff', 'left', 'alphabetic', '1px');
      bars(k, 28, y + 116, 150, 2, dark ? 'rgba(30,30,33,.25)' : 'rgba(255,255,255,.45)');
    });
    k.restore();
    // 右の列: 2,1,0 の順に上から並び（組 i は下から i 番目）、下へ流れる
    k.save(); k.beginPath(); k.rect(HW, 0, HW, VH); k.clip();
    SEC.forEach(([col, no, word, v], i) => {
      const y = (s - i) * VH;
      if (y > VH || y < -VH) return;
      art(k, HW, y, HW, VH, v, 0);
    });
    k.restore();
    // 真ん中の境目と、組の目印
    k.fillStyle = '#fff'; k.fillRect(HW - 1, 0, 2, VH);
    for (let i = 0; i < 3; i++) { const on = clamp(1 - Math.abs(s - i) * 1.5); k.fillStyle = `rgba(255,255,255,${0.45 + 0.55 * on})`; k.beginPath(); k.arc(HW, 58 + i * 27, 3.5 + 2 * on, 0, 7); k.fill(); k.strokeStyle = 'rgba(30,30,33,.25)'; k.lineWidth = 1; k.stroke(); }
    // 矢印（動いている間だけ。左右は逆向き）
    const ds = steps([[0.35, 0], [1.15, 1], [1.95, 1], [2.75, 2], [3.6, 2], [4.45, 0]], u + 0.02) - s;
    if (Math.abs(ds) > 1e-4) {
      const d = ds > 0 ? 1 : -1;
      k.fillStyle = '#fff';
      const tri = (x, y, up) => { k.beginPath(); k.moveTo(x, y + (up ? -7 : 7)); k.lineTo(x - 8, y + (up ? 5 : -5)); k.lineTo(x + 8, y + (up ? 5 : -5)); k.fill(); };
      tri(HW - 22, 146, d > 0); tri(HW + 22, 146, d < 0);
    }
  });
};
