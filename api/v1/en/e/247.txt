// Cover Flow / カバーフロー
// Pack: lp3 (Landing page & web effects, part 3) · element 07 · Category: Scroll & drag effects
// What happens: Square covers line up on a dark background; all but the center one are turned and overlapping. Each step swings the next cover round to the front and the title changes. Reflections on the floor.
// Motion spec:
//   - A 511 × 170 viewport: covers of 88 × 88 on a light background (white → #F1F0EE). One pass takes 4.7 s (twice per 9.4 s pack cycle).
//   - Front index p: 0→1 over 0.25–0.75 s, 1→2 over 1.05–1.55 s, 2→3 over 1.85–2.35 s, 3→0 over 3.0–4.3 s (cubic ease-in-out).
//   - Cover i has rel = i − p. Horizontal offset: rel × 74 px while |rel| < 1, then 74 + (|rel| − 1) × 30 px. Rotation = clamp(|rel|) × 62° (mirrored on each side); turned covers are washed out with white (up to 45%).
//   - Drawn far to near, the front one last. A flipped reflection (opacity 0.22) fades into the floor gradient. The front title shows only while p is near a whole number.
//   - Unlike #058 Carousel (slide sideways one by one), the covers on both sides are turned away and stacked like a 3D shelf.
// When to use it: Use it to browse albums, books or packaging, anything with a cover people pick by sight.
// Tags: cover-flow, carousel, 3d, gallery, album, reflection, coverflow
// Size: 539x208 px · Cycle: 9.4 s
// Needs: kit lp3 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp3.js
// Live demo: https://showheyohtaki.github.io/motion-prompt/#247
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 07
// 07 カバーフロー（正面に1枚、左右のジャケットは斜めに傾いて重なる。送ると次の1枚が回り込んで正面に来る。床に映り込み）
CELLS[6] = (k, w, h, u) => {
  u = snapU(u);
  frame(k, 0);
  const p = steps([[0.25, 0], [0.75, 1], [1.05, 1], [1.55, 2], [1.85, 2], [2.35, 3], [3.0, 3], [4.3, 0]], u);   // 正面にある番号（0〜3）
  const NAMES = ['Sunset Club', 'Lemon Tree', 'Mint Night', 'Pink Noise', 'Blue Hour', 'Deep Sea'];
  view(k, () => {
    const bg = k.createLinearGradient(0, 0, 0, VH); bg.addColorStop(0, '#FFFFFF'); bg.addColorStop(1, C.desk); k.fillStyle = bg; k.fillRect(0, 0, VW, VH);
    const CX = 255, CY = 70, S = 88;
    const items = [];
    for (let i = Math.floor(p) - 5; i <= Math.ceil(p) + 5; i++) items.push(i);
    items.sort((a, b) => Math.abs(b - p) - Math.abs(a - p));            // 遠い物から描く
    for (const i of items) {
      const rel = i - p, ar = Math.abs(rel), sg = Math.sign(rel);
      const x = CX + (ar < 1 ? rel * 74 : sg * (74 + (ar - 1) * 30));    // 正面から1つ目は74px、その先は30pxずつ
      const a = clamp(ar) * 62 * Math.PI / 180 * sg;                     // 回転（最大62°）
      if (Math.abs(x - CX) > 300) continue;
      const sx = Math.cos(a), skw = Math.sin(a) * 0.28;
      const draw = (flip) => {
        k.save(); k.translate(x, CY + (flip ? S : 0)); k.transform(sx, -skw, 0, flip ? -1 : 1, 0, 0);
        art(k, -S / 2, flip ? -S : -S / 2, S, S, mod(i, 6), 3);
        if (!flip) { txt(k, String(mod(i, 6) + 1).padStart(2, '0'), -S / 2 + 8, -S / 2 + 16, `800 10px ${MONO}`, rgba(C.ink, 0.55)); }
        k.restore();
      };
      k.globalAlpha = 0.22; draw(true); k.globalAlpha = 1;
      draw(false);
      k.fillStyle = `rgba(255,255,255,${0.45 * clamp(ar)})`;                // 横を向いた物は暗く
      k.save(); k.translate(x, CY); k.transform(sx, -skw, 0, 1, 0, 0); k.fillRect(-S / 2, -S / 2, S, S); k.restore();
    }
    // 映り込みを床でぼかす
    const fl = k.createLinearGradient(0, CY + S / 2, 0, VH); fl.addColorStop(0, rgba(C.desk, 0.35)); fl.addColorStop(0.6, rgba(C.desk, 1)); k.fillStyle = fl; k.fillRect(0, CY + S / 2, VW, VH);
    // 曲名（正面の1枚）
    const near = Math.round(p), la = clamp(1 - Math.abs(p - near) * 3);
    txt(k, NAMES[mod(near, 6)], CX, 140, `800 13px ${EN}`, rgba(C.ink, la), 'center');
    txt(k, 'LP3 Records', CX, 156, `500 10px ${EN}`, rgba(C.sub, la), 'center');
    // 送りのつまみ
    k.fillStyle = C.faint; rr(k, 150, 162, 210, 4, 2); k.fill();
    k.fillStyle = C.blue; rr(k, 150 + 210 * p / 5, 160, 36, 8, 4); k.fill();
  });
};
