// Infinite canvas / 無限キャンバス
// Pack: lp3 (Landing page & web effects, part 3) · element 08 · Category: Scroll & drag effects
// What happens: Project tiles sit on a grid; grabbing and dragging moves the whole wall up-left or left. However far it goes there is no edge; the same arrangement keeps coming back.
// Motion spec:
//   - A 511 × 170 viewport. Tile grid: 96 across, 84 down, every other row shifted by 48 px; tiles come in three sizes (74 × 58, 64 × 52, 70 × 46). One pass takes 4.7 s (twice per 9.4 s pack cycle).
//   - Two drags: 288 px left and 168 px up over 0.3–1.9 s, then 288 px left over 2.5–4.2 s (bulging 14 px vertically on the way); cubic ease-in-out.
//   - Tile art and size repeat every 6 × 2 cells. One pass moves exactly a multiple of that (576 × 168), so there is no seam.
//   - While grabbing, the round handle shows a ring; after letting go it moves back to where the next drag starts. A 24 px grid moves along behind.
//   - #091 Infinite scroll loads more content downward; an infinite canvas lets you move around a 2D surface in any direction.
// When to use it: Use it for portfolios or exhibitions with many pieces, when browsing should feel like exploring.
// Tags: infinite-canvas, drag, pan, grid, portfolio, explore
// Size: 539x208 px · Cycle: 9.4 s
// Needs: kit lp3 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp3.js
// Live demo: https://showheyohtaki.github.io/motion-prompt/#248
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 08
// 08 無限キャンバス（上下左右に途切れなく続く作品の壁を、つかんでドラッグして見て回る。端が来ない）
CELLS[7] = (k, w, h, u) => {
  u = snapU(u);
  frame(k, 0);
  // ドラッグは2回。1回目は左上へ、2回目は左へ。1周で横に6枚・縦に2段ぶん進む（タイルは 6×2 で繰り返すので継ぎ目なし）
  const TX = 96, TY = 84;
  const d1 = eIO(P(u, 0.3, 1.9)), d2 = eIO(P(u, 2.5, 4.2));
  const ox = -(TX * 3) * d1 - (TX * 3) * d2, oy = -(TY * 2) * d1 + 0 * d2 + 14 * Math.sin(Math.PI * d2);
  view(k, () => {
    k.fillStyle = C.soft; k.fillRect(0, 0, VW, VH);
    // うすい方眼
    k.strokeStyle = rgba(C.blue, 0.08); k.lineWidth = 1;
    for (let x = mod(ox, 24); x < VW; x += 24) { k.beginPath(); k.moveTo(x, 0); k.lineTo(x, VH); k.stroke(); }
    for (let y = mod(oy, 24); y < VH; y += 24) { k.beginPath(); k.moveTo(0, y); k.lineTo(VW, y); k.stroke(); }
    const i0 = Math.floor(-ox / TX) - 1, j0 = Math.floor(-oy / TY) - 1;
    for (let j = j0; j < j0 + 4; j++) for (let i = i0; i < i0 + 7; i++) {
      const x = ox + i * TX + (mod(j, 2) ? 48 : 0) + 10, y = oy + j * TY + 12;
      const v = mod(i * 5 + j * 3, 6), sz = mod(i + j * 3, 3);
      const tw = [74, 64, 70][sz], th = [58, 52, 46][sz];
      k.save(); k.shadowColor = 'rgba(30,30,33,.14)'; k.shadowBlur = 8; k.shadowOffsetY = 3; art(k, x, y, tw, th, v, 6); k.restore();
    }
    // つかむ印: ドラッグ中は押した輪、間は離して元の位置へ戻る
    const on = (u > 0.3 && u < 1.9) || (u > 2.5 && u < 4.2);
    const gx = on ? (u < 2 ? 380 - 120 * d1 : 400 - 150 * d2) : (u >= 1.9 && u < 2.5 ? lerp(260, 400, eIO(P(u, 1.9, 2.5))) : u >= 4.2 ? lerp(250, 380, eIO(P(u, 4.2, 4.7))) : 380);
    const gy = on ? (u < 2 ? 120 - 60 * d1 : 90 + 14 * Math.sin(Math.PI * d2)) : (u >= 1.9 && u < 2.5 ? lerp(60, 90, eIO(P(u, 1.9, 2.5))) : u >= 4.2 ? lerp(90, 120, eIO(P(u, 4.2, 4.7))) : 120);
    grabber(k, gx, gy, on);
    // 右上の札
    k.fillStyle = 'rgba(30,30,33,.82)'; rr(k, 392, 10, 106, 20, 10); k.fill();
    txt(k, 'DRAG TO EXPLORE', 445, 20.5, `700 8.5px ${EN}`, '#fff', 'center', 'middle', '1px');
  });
};
