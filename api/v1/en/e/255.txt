// Split-flap display / パタパタ表示
// Pack: lp3 (Landing page & web effects, part 3) · element 15 · Category: Pointer interactions
// What happens: Eleven letter tiles on a black board split and flip, passing through a few letters on the way, to change TOKYO 09:30 → PARIS 11:05 → SEOUL 13:40.
// Motion spec:
//   - A 511 × 170 viewport: eleven 34 × 54 letter tiles (radius 5) 4 px apart. One pass takes 4.7 s (twice per 9.4 s pack cycle).
//   - Changes start at 0.5, 2.0 and 3.5 s (TOKYO 09:30 → PARIS 11:05 → SEOUL 13:40 → back). Letter i starts i × 0.045 s later.
//   - Each flip takes 0.09 s. Letters that stay the same do not move; others flip 3 to 5 times through intermediate letters and stop on the new one.
//   - One flip f (0–1): behind, the top half shows the next letter and the bottom half the current one. While f < 0.5 the current top half folds down by scaling vertically around the middle (darkening); from 0.5 the next bottom half unfolds downward.
//   - #156 Text roll on hover rolls letters over and #129 Text scramble swaps random letters; a split-flap display shows the split tiles physically falling.
// When to use it: Use it for travel, events or countdowns, to change places, times or numbers with a mechanical feel.
// Tags: split-flap, solari, flip, departure-board, text, retro
// Size: 539x208 px · Cycle: 9.4 s
// Needs: kit lp3 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp3.js
// Live demo: https://showheyohtaki.github.io/motion-prompt/#255
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 15
// 15 パタパタ表示（駅の発車標のように、1文字ずつの板が上下に割れてめくれ、何枚か途中の文字を経て次の文字に止まる。左から少しずつずれて始まる）
CELLS[14] = (k, w, h, u) => {
  u = snapU(u);
  frame(k, 1);
  const WORDS = ['TOKYO 09:30', 'PARIS 11:05', 'SEOUL 13:40'];
  const START = [0.5, 2.0, 3.5];                                        // 0→1, 1→2, 2→0 の切り替え開始
  const CH = ' ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789:';
  const FL = 0.09, STAG = 0.045;                                        // 1回めくるのに0.09秒、1文字ごとに0.045秒ずつ遅れる
  // 文字 i の、時刻 u での [今の文字, 次の文字, めくれ具合 0〜1]
  const state = i => {
    let seg = 2; for (let s = 0; s < 3; s++) if (u >= START[s] + i * STAG) seg = s;
    if (u < START[0] + i * STAG) seg = 2;
    const from = WORDS[seg][i], to = WORDS[(seg + 1) % 3][i];
    const t = u - START[seg] - i * STAG + (u < START[0] + i * STAG ? CYC : 0);
    if (from === to) return [to, to, 0];
    const nf = 3 + mod(i * 5 + seg, 3);                                // 途中の文字を含めて3〜5回めくる
    const seq = [from]; for (let j = 1; j < nf; j++) seq.push(CH[mod(CH.indexOf(from) + j * 7 + i, CH.length)]); seq.push(to);
    const step = Math.floor(t / FL);
    if (step >= nf) return [to, to, 0];
    return [seq[step], seq[step + 1], (t - step * FL) / FL];
  };
  view(k, () => {
    k.fillStyle = C.blueBg; k.fillRect(0, 0, VW, VH);
    k.fillStyle = C.ink; rr(k, 24, 30, 463, 110, 14); k.fill();
    txt(k, 'DEPARTURES', 44, 52, `800 10px ${EN}`, C.blue, 'left', 'middle', '3px');
    txt(k, '出発', 467, 52, `800 11px ${JP}`, 'rgba(255,255,255,.7)', 'right', 'middle');
    const N = 11, FW = 34, FH = 54, G = 4, x0 = (VW - (N * FW + (N - 1) * G)) / 2, y0 = 66, mid = y0 + FH / 2;
    const glyph = (c, x, i) => txt(k, c, x + FW / 2, mid + 1.5, `800 30px ${MONO}`, i >= 6 ? C.blueL : '#fff', 'center', 'middle');
    const half = (c, x, i, top, sy = 1) => {       // 上半分 or 下半分（sy で真ん中を軸に縮める）
      k.save(); k.beginPath(); k.rect(x, top ? y0 : mid, FW, FH / 2); k.clip();
      k.translate(0, mid); k.scale(1, sy); k.translate(0, -mid);
      k.fillStyle = C.mark; rr(k, x, y0, FW, FH, 5); k.fill(); glyph(c, x, i);
      if (sy < 1) { k.fillStyle = `rgba(0,0,0,${0.35 * (1 - sy)})`; k.fillRect(x, y0, FW, FH); }
      k.restore();
    };
    for (let i = 0; i < N; i++) {
      const x = x0 + i * (FW + G), [a, b, f] = state(i);
      if (f <= 0) { half(a, x, i, true); half(a, x, i, false); }
      else {
        half(b, x, i, true); half(a, x, i, false);                     // うしろ: 上は次の字、下は今の字
        if (f < 0.5) half(a, x, i, true, 1 - f * 2);                   // 今の字の上半分が倒れてくる
        else half(b, x, i, false, f * 2 - 1);                          // 次の字の下半分が下りてくる
      }
      k.fillStyle = C.ink; k.fillRect(x, mid - 0.75, FW, 1.5);
    }
    bars(k, 44, 128, 120, 1, 'rgba(255,255,255,.18)', 4);
  });
};
