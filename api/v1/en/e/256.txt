// FLIP animation / FLIPアニメーション
// Pack: lp3 (Landing page & web effects, part 3) · element 16 · Category: Pointer interactions
// What happens: Pressing "写真" shrinks the illustration cards away and the remaining photo cards slide from their old spots into the gaps. "すべて" plays it back in reverse.
// Motion spec:
//   - A 511 × 170 viewport: 106 × 52 cards (radius 7) in 4 columns × 2 rows, 118 across and 62 down, with three filter buttons on top. One pass takes 4.7 s (twice per 9.4 s pack cycle).
//   - At 0.6 s "写真" is pressed: over 0.6–0.85 s the four illustrations shrink 1 → 0.75 and fade 1 → 0 (cubic ease-out); over 0.75–1.4 s the four photos move from their old spots to the packed ones (cubic ease-in-out).
//   - At 2.6 s "すべて" is pressed: the photos return over 2.6–3.2 s and the illustrations reappear over 3.0–3.35 s. The count reads 8 件 ↔ 4 件.
//   - While cards move, a dashed outline stays at each old spot (First) to show where they came from.
//   - FLIP = First (old position), Last (new position), Invert (cancel the difference with a transform), Play (animate the transform back to 0): the way to animate a layout change by its position difference.
// When to use it: Use it whenever filtering, sorting, adding or removing changes a layout, so people can see where each item went.
// Tags: flip, layout-animation, filter, shuffle, grid, transition
// Size: 539x208 px · Cycle: 9.4 s
// Needs: kit lp3 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp3.js
// Live demo: https://showheyohtaki.github.io/motion-prompt/#256
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 16
// 16 FLIPアニメーション（絞り込みのボタンを押すと、外れたカードが縮んで消え、残ったカードが元の場所から新しい場所へ滑って並び直す）
CELLS[15] = (k, w, h, u) => {
  u = snapU(u);
  frame(k, 1);
  const TYPE = [0, 1, 0, 1, 1, 0, 0, 1], V = [1, 0, 4, 2, 3, 1, 4, 5];   // 0＝写真（山の絵）、1＝イラスト（丸・しま）
  const slot = n => [24 + (n % 4) * 118, 44 + Math.floor(n / 4) * 62];
  const after = []; let c = 0; TYPE.forEach((t, i) => { after[i] = t === 0 ? c++ : -1; });
  // 写真で絞る: 0.6 押す → 0.6〜0.85 イラストが消える → 0.75〜1.4 写真が滑る。すべてに戻す: 2.6 押す → 2.6〜3.2 戻る → 3.0〜3.35 イラストが出る
  const mv = eIO(P(u, 0.75, 1.4)) * (1 - eIO(P(u, 2.6, 3.2)));
  const gone = eOut(P(u, 0.6, 0.85)) * (1 - eOut(P(u, 3.0, 3.35)));
  const [cx, cy] = path([[0, 430, 140], [0.45, 130, 26], [1.4, 130, 26], [2.0, 320, 150], [2.45, 62, 26], [3.4, 62, 26], [4.4, 430, 140]], u);
  const press = Math.max(clamp(1 - Math.abs(u - 0.6) / 0.08), clamp(1 - Math.abs(u - 2.6) / 0.08));
  const sel = u >= 0.6 && u < 2.6 ? 1 : 0;
  view(k, () => {
    k.fillStyle = '#fff'; k.fillRect(0, 0, VW, VH);
    ['すべて', '写真', 'イラスト'].forEach((s, i) => {
      const x = 24 + i * 72, on = i === (sel ? 1 : 0);
      k.fillStyle = on ? C.ink : C.soft; rr(k, x, 12, 64, 22, 11); k.fill();
      txt(k, s, x + 32, 23.5, `700 10.5px ${JP}`, on ? '#fff' : C.ink, 'center', 'middle');
    });
    txt(k, `${sel ? 4 : 8} 件`, 487, 23.5, `600 10px ${JP}`, C.sub, 'right', 'middle');
    for (let i = 0; i < 8; i++) {
      const [x0, y0] = slot(i);
      if (TYPE[i] === 1) {
        if (gone >= 1) continue;
        const s = 1 - 0.25 * gone; k.save(); k.globalAlpha = 1 - gone; k.translate(x0 + 53, y0 + 26); k.scale(s, s); art(k, -53, -26, 106, 52, V[i], 7); k.restore();
        continue;
      }
      const [x1, y1] = slot(after[i]), x = lerp(x0, x1, mv), y = lerp(y0, y1, mv);
      // First（元の場所）の点線の枠: 動いている間だけ
      if (mv > 0.02 && mv < 0.98 && (x0 !== x1 || y0 !== y1)) { k.save(); k.setLineDash([4, 4]); k.strokeStyle = rgba(C.blue, 0.45); k.lineWidth = 1.2; rr(k, x0 + 0.5, y0 + 0.5, 105, 51, 7); k.stroke(); k.restore(); }
      k.save(); k.shadowColor = 'rgba(30,30,33,.16)'; k.shadowBlur = 8; k.shadowOffsetY = 3; art(k, x, y, 106, 52, V[i], 7); k.restore();
    }
    cursor(k, cx, cy, press);
  });
};
