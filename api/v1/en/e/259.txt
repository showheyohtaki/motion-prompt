// Liquid metal / リキッドメタル
// Pack: lp3 (Landing page & web effects, part 3) · element 19 · Category: Backgrounds & textures
// What happens: Inside the big word FLOW and a sphere on a light page, black, blue and white reflections keep flowing and rippling.
// Motion spec:
//   - A 511 × 170 viewport on a light gradient (white → #F1F0EE), with the word FLOW (84 px) and a sphere (radius 52). One pass takes 4.7 s (twice per 9.4 s pack cycle).
//   - Pattern value v = sin(0.03x + 2.3·sin(0.045y + a) + a) + 0.55·sin(0.08y − 0.018x + 2a) + 0.3·sin(0.05(x+y) − a), with a = 2πt/4.7.
//   - Color maps v × 0.22 + y × 0.002 onto a repeating band: black #1E1E21 → blue #2F6BEB → #C9D8FB → white → #9DB8F6 → white → #E4ECFE → #2F6BEB → #1E1E21.
//   - Computed on a 128 × 43 grid, stretched and cut to the shapes of the word and the sphere. The sphere gets a white highlight at the top left and shade at the bottom right; a blue glow (opacity 0.25, blur 14) around everything.
//   - #022 Light sweep is one band of light crossing; liquid metal is the whole surface's reflection warping and moving.
// When to use it: Use it for futuristic, tactile headings and logos on AI, tech or music sites.
// Tags: liquid-metal, chrome, metallic, shader, reflection, typography
// Size: 539x208 px · Cycle: 9.4 s
// Needs: kit lp3 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp3.js
// Live demo: https://showheyohtaki.github.io/motion-prompt/#259
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 19
// 19 リキッドメタル（見出しの文字と球の中を、溶けた金属のような映り込みがうねりながら流れ続ける）
CELLS[18] = (k, w, h, u) => {
  u = snapU(u);
  frame(k, 2);
  // 金属の色の並び（黒→青→薄い青→白→中間の青→白→ごく薄い青→青→黒）
  const RAMP = [C.ink, C.blue, C.blueL, '#FFFFFF', C.blueM, '#FFFFFF', C.blueXL, C.blue, C.ink].map(hex);
  const ramp = t => { t = mod(t, 1) * (RAMP.length - 1); const i = Math.floor(t), f = t - i, a = RAMP[i], b = RAMP[Math.min(i + 1, RAMP.length - 1)];
    return `rgb(${lerp(a[0], b[0], f) | 0},${lerp(a[1], b[1], f) | 0},${lerp(a[2], b[2], f) | 0})`; };
  view(k, () => {
    const bg = k.createLinearGradient(0, 0, 0, VH); bg.addColorStop(0, '#FFFFFF'); bg.addColorStop(1, C.desk); k.fillStyle = bg; k.fillRect(0, 0, VW, VH);
    const ph = W1 * u;
    const [c, g] = layer(k, 'metal', VW, VH);
    field(g, 'metal-f', 0, 0, VW, VH, 128, 43, (i, j) => {
      const x = i * 4, y = j * 4;
      const v = Math.sin(x * 0.03 + 2.3 * Math.sin(y * 0.045 + ph) + ph) + 0.55 * Math.sin(y * 0.08 - x * 0.018 + 2 * ph) + 0.3 * Math.sin((x + y) * 0.05 - ph);
      return ramp(v * 0.22 + y * 0.002);
    });
    // 形だけ残す（文字と球を1枚の型にまとめてから切り抜く）
    const [mc, mg] = layer(k, 'metal-m', VW, VH);
    txt(mg, 'FLOW', 30, 112, `900 84px ${EN}`, '#000', 'left', 'alphabetic', '2px');
    mg.beginPath(); mg.arc(414, 80, 52, 0, 7); mg.fill(); mg.restore();
    g.globalCompositeOperation = 'destination-in'; g.drawImage(mc, 0, 0, VW, VH);
    g.globalCompositeOperation = 'source-over';
    // 球の立体感
    const sg = g.createRadialGradient(396, 60, 6, 414, 80, 56); sg.addColorStop(0, 'rgba(255,255,255,.55)'); sg.addColorStop(0.55, 'rgba(255,255,255,0)'); sg.addColorStop(1, 'rgba(30,30,33,.55)');
    g.fillStyle = sg; g.beginPath(); g.arc(414, 80, 52, 0, 7); g.fill();
    g.restore();
    k.save(); k.shadowColor = rgba(C.blue, 0.25); k.shadowBlur = 14; k.drawImage(c, 0, 0, VW, VH); k.restore();
    txt(k, 'Liquid metal type & objects', 32, 140, `600 11px ${EN}`, C.sub);
    k.fillStyle = 'rgba(30,30,33,.08)'; k.beginPath(); k.ellipse(414, 150, 46, 6, 0, 0, 7); k.fill();
  });
};
