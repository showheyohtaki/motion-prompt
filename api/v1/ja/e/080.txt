// マーカーアニメーション / Highlighter animation
// パック: lp (LP・Webサイトの演出) · element 08 · カテゴリ: スクロールで起きる演出
// 何が起きるか: スクロールで文章が画面に入ると、大事な言葉の下半分に黄色い蛍光ペンの線が左から引かれる。
// 動きの仕様:
//   - スクロール量: 0.35〜1.0秒で 0→90px、1.6〜2.2秒で 90→180px、3.3〜4.4秒で 180→0（easeInOut 3乗）
//   - 文章は2行（20px・太字）。ページ上の 190px と 290px にあり、最初は画面の下に隠れている
//   - 行の上端が判定線（表示域の高さの80%＝134px）より上に来た時刻に、強調する言葉の下に線を引き始める
//   - 線: 黄色 #FFE36B、高さ11（文字の下寄り。ベースラインの8px上〜3px下）。幅 0→言葉の幅 を0.8秒で左から（easeOut 3乗）。文字の後ろに描く
//   - 画面の下へ出た行は、見えない所で線を消して元に戻す
// 使いどころ: LPの本文で、いちばん伝えたい一言を読んでいる瞬間に目立たせたいとき。
// タグ: highlighter, marker, text-highlight, emphasis, scroll-reveal, copy, underline
// サイズ: 539x208 px · 1周: 4.8 s
// Needs: kit lp -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp.js
// 動く見本: https://showheyohtaki.github.io/motion-prompt/#080
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 08
// 08 マーカーアニメーション（大事な言葉が画面に入ると、蛍光ペンの線が左から引かれる）
CELLS[7] = (k, w, h, u) => {
  frame(k);
  const SY = t => steps([[0.35, 0], [1.0, 90], [1.6, 90], [2.2, 180], [3.3, 180], [4.4, 0]], t), y = SY(u);
  const LINE = VH * 0.8, DUR = 0.8, MARK = '#FFE36B', F = `800 20px ${JP}`;
  const rows = [[190, ['毎日のごはんが', 'たった10分で', '作れる。']], [290, ['はじめてでも', '失敗しないレシピ', 'だけ。']]];
  view(k, () => {
    k.save(); k.translate(0, -y);
    txt(k, '忙しい人のための', 70, 50, `700 14px ${JP}`, C.sub); txt(k, '台所の新しい相棒', 70, 80, `800 24px ${JP}`, C.ink);
    bars(k, 70, 96, 330, 4, C.faint, 6, 12, 0.8);
    for (const [base, parts] of rows) {
      k.font = F; const ws = parts.map(s => k.measureText(mpText(s)).width); let x = (VW - ws.reduce((a, b) => a + b, 0)) / 2;
      const t0 = enteredAt(SY, base - 20, LINE, u), q = t0 === null ? 0 : eOut(P(u, t0, t0 + DUR));
      if (q > 0) { k.fillStyle = MARK; k.fillRect(x + ws[0], base - 8, ws[1] * q, 11); }
      parts.forEach((s, i) => { txt(k, s, x, base, F, i === 1 ? C.ink : '#4A5368'); x += ws[i]; });
      bars(k, 110, base + 22, 291, 2, C.faint, 6, 12, 0.7);
    }
    k.restore();
    sbar(k, y / 180, 0.5);
  });
};
