// ピラミッド図 / Pyramid diagram
// パック: diagram (資料で使う図解・グラフの名前) · element 04 · カテゴリ: 整理する図
// 何が起きるか: 4段の三角形が、下の段から順に上から落ちてきて積み上がる。一番上の段だけ濃い青。
// 動きの仕様:
//   - 形: 頂点y=16・底辺y=192・底辺の幅336pxの三角形を4段に分ける（段の間6px）
//   - 色: 下から薄いグレー→薄い青→中くらいの青→濃い青
//   - 動き: 下の段から0.32秒おき（0.4・0.72・1.04・1.36秒）に、34px上から0.4秒で落ちてきながらフェードイン（easeOut）
//   - 終わり: 4.1〜4.5秒で図全体がフェードアウトし、空のまま4.7秒で1周
// 使いどころ: 土台から上へ積み上がる段階や、上ほど少ない階層を見せるとき。
// タグ: pyramid, pyramid diagram, hierarchy, layers, levels, triangle, stack
// サイズ: 539x208 px · 1周: 4.7 s
// Needs: kit diagram -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/diagram.js
// 動く見本: https://showheyohtaki.github.io/motion-prompt/#220
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 04
// 04 ピラミッド図: a triangle in four layers, stacked from the bottom up; the top layer is blue
function pyramid(k, w, h, u) {
  bg(k, w, h);
  k.globalAlpha = fade(u);
  const top = 16, base = 192, hwB = 168, n = 4, gp = 6, cx = w / 2;
  const th = (base - top - gp * (n - 1)) / n, half = y => hwB * (y - top) / (base - top);
  const cols = [C.g3, C.blueL, C.blueM, C.blue];
  for (let i = 0; i < n; i++) {                       // i = 0 bottom
    const p = easeOut(P(u, 0.4 + i * 0.32, 0.8 + i * 0.32));
    if (p <= 0) continue;
    const yb = base - i * (th + gp), yt = yb - th, dy = (1 - p) * -34;
    k.save(); k.globalAlpha *= clamp(p * 1.6); k.translate(0, dy);
    k.fillStyle = cols[i]; k.beginPath();
    if (i === n - 1) { k.moveTo(cx, top); }
    else { k.moveTo(cx - half(yt), yt); k.lineTo(cx + half(yt), yt); }
    k.lineTo(cx + half(yb), yb); k.lineTo(cx - half(yb), yb); k.closePath(); k.fill();
    k.restore();
  }
  k.globalAlpha = 1;
}
CELLS[3] = pyramid;
