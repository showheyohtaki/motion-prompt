// ツリーマップ / Treemap
// パック: diagram (資料で使う図解・グラフの名前) · element 21 · カテゴリ: 流れと全体を見せる図
// 何が起きるか: 点線の外枠の中が、大きい四角から順に、大きさの違う7つの四角で埋まっていく。
// 動きの仕様:
//   - 外枠: 439×172pxの点線（6px・間6px）。1周ずっと出したまま
//   - 四角: 7つ（角丸5px）。左の大きいA（206×172px・濃い青）→B（140×96px）→C→D→小さい3つの順。色は濃い青→中くらいの青→薄い青→グレー
//   - 動き: 0.4秒から0.22秒おきに、0.4秒で0.55倍から弾んで広がる（easeOutBack）＋フェードイン
//   - 文字: A〜Dの左上に記号（Aは24pxの白、ほかは20px）
//   - 終わり: 4.1〜4.5秒で図全体がフェードアウトし、空のまま4.7秒で1周
// 使いどころ: 全体の中で、どれがどのくらいの大きさを占めるかを一目で見せるとき。
// タグ: treemap, tree map, part to whole, breakdown, rectangles, hierarchy, share
// サイズ: 539x208 px · 1周: 4.7 s
// Needs: kit diagram -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/diagram.js
// 動く見本: https://showheyohtaki.github.io/motion-prompt/#237
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 21
// 21 ツリーマップ: one outer rectangle gets filled by rectangles of different sizes, biggest first
function treemap(k, w, h, u) {
  bg(k, w, h);
  const f = fade(u);
  const X0 = 50, Y0 = 18, X1 = 489, Y1 = 190;
  k.strokeStyle = C.g3; k.lineWidth = 2; k.setLineDash([6, 6]);
  rr(k, X0, Y0, X1 - X0, Y1 - Y0, 6); k.stroke(); k.setLineDash([]);
  const RS = [                                  // x, y, w, h, color, label
    [50, 18, 206, 172, C.blue, 'A'],
    [260, 18, 140, 96, C.blueM, 'B'],
    [404, 18, 85, 96, C.blueL, 'C'],
    [260, 118, 98, 72, C.blueL, 'D'],
    [362, 118, 70, 72, C.g3, ''],
    [436, 118, 53, 34, C.g3, ''],
    [436, 156, 53, 34, C.g4, ''],
  ];
  RS.forEach(([x, y, ww, hh, col, lb], i) => {
    const p = P(u, 0.4 + i * 0.22, 0.8 + i * 0.22);
    if (p <= 0) return;
    const s = lerp(0.55, 1, easeOutBack(p));
    k.save(); k.globalAlpha = f * Math.min(1, p * 2.5);
    k.translate(x + ww / 2, y + hh / 2); k.scale(s, s);
    k.fillStyle = col; rr(k, -ww / 2, -hh / 2, ww, hh, 5); k.fill();
    if (lb) txt(k, lb, -ww / 2 + 12, -hh / 2 + 26, `700 ${i === 0 ? 24 : 20}px ${FONT}`, i === 0 ? '#FFFFFF' : C.mark, 'left');
    k.restore();
  });
}
CELLS[20] = treemap;
