// 下から現れるフッター / Reveal footer
// パック: lp3 (LP・Webサイトの演出 第3弾) · element 06 · カテゴリ: スクロール・ドラッグで見せる
// 何が起きるか: 最後までスクロールすると、白い本文が影を付けて上がっていき、その下に固定されていた青いフッターが現れる。フッターの中身は少し下から上がってくる。
// 動きの仕様:
//   - 表示域 511×170。フッター（高さ130・青 #2F6BEB）は画面の下に固定。4.7秒でひと回り（パックの周期9.4秒で2回）
//   - スクロール: 0.35〜2.0秒で 0→130px、3.0〜4.35秒で 130→0（easeInOut 3乗）。本文は白でフッターの上に重なり、スクロール分だけ上がる
//   - 本文の下の端: 角丸 0→14（q×3 で効く）、影 ぼかし18・下へ6・不透明度 0→0.35（q×4）。q＝スクロール量 / 130
//   - フッターの中身は 26×(1−q) px 下から元の位置へ。見出し2行（28px、2行目は薄い青 #C9D8FB）、リンク4つ、白い丸に青い矢印のボタン
//   - CSS なら フッターを position: sticky; bottom: 0（または fixed）にして、本文に背景色と z-index を付けて前に出す
// 使いどころ: お問い合わせや採用への誘導を、ページの最後に印象よく見せたいときに。
// タグ: footer, reveal, sticky-footer, curtain, scroll, parallax
// サイズ: 539x208 px · 1周: 9.4 s
// Needs: kit lp3 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp3.js
// 動く見本: https://showheyohtaki.github.io/motion-prompt/#246
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 06
// 06 下から現れるフッター（フッターは画面の下に固定され、本文の下に隠れている。最後までスクロールすると本文がめくれるように上がり、フッターが現れる）
CELLS[5] = (k, w, h, u) => {
  u = snapU(u);
  frame(k, 0);
  const y = steps([[0.35, 0], [2.0, 130], [3.0, 130], [4.35, 0]], u);   // スクロール量。130px でフッターが全部見える
  const q = y / 130;
  view(k, () => {
    // フッター（固定。中身は少し下から上がってくる）
    const FY = VH - 130;
    k.fillStyle = C.blue; k.fillRect(0, FY, VW, 130);
    const fy = FY + 26 * (1 - q);
    txt(k, "Let's make", 28, fy + 48, `900 28px ${EN}`, '#fff');
    txt(k, 'something.', 28, fy + 80, `900 28px ${EN}`, C.blueL);
    [['Work', 'About'], ['Blog', 'Contact']].forEach((col, i) => col.forEach((s, j) => txt(k, s, 290 + i * 80, fy + 44 + j * 22, `600 12px ${EN}`, 'rgba(255,255,255,.85)')));
    k.fillStyle = '#FFFFFF'; k.beginPath(); k.arc(462, fy + 66, 16, 0, 7); k.fill();
    k.strokeStyle = C.blue; k.lineWidth = 2.2; k.beginPath(); k.moveTo(456, fy + 66); k.lineTo(468, fy + 66); k.moveTo(463, fy + 61); k.lineTo(468, fy + 66); k.lineTo(463, fy + 71); k.stroke();
    // 本文（フッターの上に重なり、スクロールで上がる）
    k.save(); k.translate(0, -y);
    k.shadowColor = `rgba(30,30,33,${0.35 * Math.min(1, q * 4)})`; k.shadowBlur = 18; k.shadowOffsetY = 6;
    k.fillStyle = '#fff'; rr(k, 0, -10, VW, VH + 10, [0, 0, 14 * Math.min(1, q * 3), 14 * Math.min(1, q * 3)]); k.fill();
    k.shadowColor = 'transparent';
    txt(k, 'WORKS', 28, 40, `800 11px ${EN}`, C.blue, 'left', 'alphabetic', '2px');
    for (let i = 0; i < 3; i++) { art(k, 28 + i * 156, 54, 142, 78, i + 1, 8); bars(k, 28 + i * 156, 142, 110, 2); }
    k.restore();
    sbar(k, q, 0.55);
  });
};
