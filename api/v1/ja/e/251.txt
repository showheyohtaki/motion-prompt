// イメージトレイル / Image trail
// パック: lp3 (LP・Webサイトの演出 第3弾) · element 11 · カテゴリ: 触れると反応する
// 何が起きるか: 大きな見出しの上でカーソルを動かすと、通った所に作品の写真が次々に置かれていき、古い物から縮んで消える。
// 動きの仕様:
//   - 表示域 511×170。見出し PORTFOLIO（54px）の上で。4.7秒でひと回り（パックの周期9.4秒で2回）
//   - カーソルは8の字: x = 255 + 185·sin(2πt/4.7)、y = 88 + 46·sin(4πt/4.7 + 0.6)
//   - 0.1175秒（4.7秒の1/40）ごとに、その時のカーソルの位置へ写真を1枚置く。写真は白い縁付き 58×72、−0.125〜0.125 rad 傾ける
//   - 置いた写真: 最初の12%（約0.1秒）で backOut で 0→1 に開き、55%〜100% で easeIn 3乗で縮んで消える。寿命0.85秒。新しい物ほど上
//   - #083 マウスストーカーは1つの丸が追いかける、#157 ホバーで画像プレビューはリストの行の写真を出す。イメージトレイルは通り道に何枚も残す
// 使いどころ: 写真家・デザイナー・スタジオのポートフォリオで、作品の多さを遊びながら見せたいときに。
// タグ: image-trail, cursor, mouse-trail, portfolio, hover, gallery
// サイズ: 539x208 px · 1周: 9.4 s
// Needs: kit lp3 -> https://showheyohtaki.github.io/motion-prompt/api/v1/kit/lp3.js
// 動く見本: https://showheyohtaki.github.io/motion-prompt/#251
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 11
// 11 イメージトレイル（カーソルを動かすと、通った所に写真が次々に置かれ、古い物から縮んで消えていく）
CELLS[10] = (k, w, h, u) => {
  u = snapU(u);
  frame(k, 1);
  const pos = t => [255 + 185 * Math.sin(W1 * t), 88 + 46 * Math.sin(2 * W1 * t + 0.6)];   // カーソルの通り道（8の字）
  const DT = CYC / 40, LIFE = 0.85;                                     // 0.1175秒ごとに1枚、0.85秒で消える
  view(k, () => {
    k.fillStyle = C.blueBg; k.fillRect(0, 0, VW, VH);
    txt(k, 'PORTFOLIO', VW / 2, 98, `900 54px ${EN}`, rgba(C.ink, 0.92), 'center', 'alphabetic', '4px');
    txt(k, 'Photo / Art / Motion — 2026', VW / 2, 124, `600 11px ${EN}`, C.sub, 'center');
    const n0 = Math.floor(u / DT);
    for (let j = Math.ceil(LIFE / DT); j >= 0; j--) {
      const n = n0 - j, t0 = n * DT, age = u - t0;
      if (age < 0 || age > LIFE) continue;
      const [x, y] = pos(t0), q = age / LIFE;
      const s = (q < 0.12 ? eBack(q / 0.12) : 1) * (1 - eIn(clamp((q - 0.55) / 0.45)));
      const m = mod(n, 40), rot = (mod(m * 37, 11) - 5) * 0.025;     // 1周で40枚。番号は40で回す（継ぎ目なし）
      k.save(); k.translate(x, y); k.rotate(rot); k.scale(s, s);
      k.shadowColor = 'rgba(30,30,33,.25)'; k.shadowBlur = 8; k.shadowOffsetY = 3;
      k.fillStyle = '#fff'; k.fillRect(-29, -36, 58, 72); k.shadowColor = 'transparent';
      art(k, -25, -32, 50, 64, mod(m, 6), 2);
      k.restore();
    }
    const [cx, cy] = pos(u); cursor(k, cx, cy);
  });
};
