// 运动拖尾 / Motion trail
// 分类: motion (动态图形) · element 14 · 子类: 运动细节
// 示例效果: 小球沿 8 字路径移动，身后留下逐渐淡去的残影。
// 效果规格:
//   - 珊瑚色小球半径 25，白色圆心半径 11；每 4.7 s 沿 8 字走一圈：x = 196·sin(a)，y = 62·sin(2a)，a 匀速增加。
//   - 共 15 个残影，每个沿路径落后 0.105 rad，约 0.079 s。
//   - 残影半径 22→6，不透明度为 0.9 × (1 − i/15) + 0.05；越早的残影越小越淡。
//   - 残影颜色从新到旧依次为珊瑚→黄→蓝。
//   - 先画最早的残影，最后画小球本体。
// 适用场景: 适合飞行物、光标和光点的速度与路径表现。
// 标签: trail, motion-trail, echo, afterimage, ghost, path, speed
// 尺寸: 539x208 px · 周期: 4.7 s
// Needs: kit motion -> https://showheyohtaki.github.io/motion-prompt/api/v1/zh-Hans/kit/motion.js
// 动态示例: https://showheyohtaki.github.io/motion-prompt/#014
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D，调用前已清空并包裹 save/restore；w、h 为尺寸，u 为周期内秒数。画面仅由 u 决定。
// 以下是示例代码，CELLS、常量和辅助函数位于 kit 中。

// @element 14
// 14 モーショントレイル（動いた跡に、残像が尾を引く）
CELLS[13] = (k, w, h, u) => {
  stage(k);
  const X = w / 2, Y = h / 2 + 2, A = 196, B = 62, th = u / CYC * Math.PI * 2;
  const pt = a => [X + A * Math.sin(a), Y + B * Math.sin(2 * a)];
  const N = 15;
  for (let i = N; i >= 1; i--) {
    const [x, y] = pt(th - i * 0.105), f = i / N;
    k.globalAlpha = 0.9 * (1 - f) + 0.05; k.fillStyle = f < 0.5 ? mix(C.coral, C.yel, f * 2) : mix(C.yel, C.blue, (f - 0.5) * 2);
    k.beginPath(); k.arc(x, y, lerp(22, 6, f), 0, 7); k.fill();
  }
  k.globalAlpha = 1; const [x, y] = pt(th);
  k.fillStyle = C.coral; k.beginPath(); k.arc(x, y, 25, 0, 7); k.fill();
  k.fillStyle = '#fff'; k.beginPath(); k.arc(x, y, 11, 0, 7); k.fill();
};
