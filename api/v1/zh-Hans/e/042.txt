// 匹配剪辑 / Match cut
// 分类: camera (镜头运动与转场) · element 18 · 子类: 转场
// 示例效果: 抛出的球在下一镜头中变为同位置同大小的太阳，用形状衔接不同场景。
// 效果规格:
//   - 衔接：2.3 s 硬切，0 帧；切换前后蓝色圆的坐标均为中心右 120 px、上 46 px，半径 17 px。
//   - 前镜头 0–2.3 s：相机固定、高 1.4 m；人物位于左 1.2 m、距 7 m，从 0.5 s 起抛球，三次缓出，约 cubic-bezier(0.33,1,0.68,1)。
//   - 球从人物头顶上方高 2.3 m 处沿弧线移至匹配点，半径 5 px→17 px，弧线向上鼓起最多 30 px。
//   - 后镜头 2.3–4.7 s：另一个无建筑的地点，相机高 1.2 m；相同位置的蓝圆成为太阳，2.4 s 内下沉 14 px。
//   - 半径 26 px 的蓝色虚线圆在切换前后各 0.7 s 内淡入淡出，标记匹配位置。
//   - 右上角显示 SHOT 1→SHOT 2；4.7 s 直接切回 SHOT 1。
// 适用场景: 适合用共同形状或动作连接时空相距较远的场景。
// 标签: match cut, graphic match, shape match, cut, transition, edit
// 尺寸: 539x208 px · 周期: 9.4 s
// Needs: kit camera -> https://showheyohtaki.github.io/motion-prompt/api/v1/zh-Hans/kit/camera.js
// 动态示例: https://showheyohtaki.github.io/motion-prompt/#042
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D，调用前已清空并包裹 save/restore；w、h 为尺寸，u 为周期内秒数。画面仅由 u 决定。
// 以下是示例代码，CELLS、常量和辅助函数位于 kit 中。

// @element 18
// 18 マッチカット（投げたボール → 同じ位置・大きさの太陽）
CELLS[17] = shot((k, w, h, u) => {
  const cut = 2.3;
  const BX = 120, BY = -46, BR = 17;
  if (u < cut) {
    const q = eOut(P(u, 0.5, cut));
    world(k, w, h, cam({ y: 1.4 }), {
      person: { x: -1.2, z: 7, armUp: q > 0.05 }, sun: null,
      after: (kk, cc) => { const s = proj(cc, -1.2, 2.3, 7); const x = lerp(s.x, BX, q), y = lerp(s.y, BY, q) - Math.sin(q * Math.PI) * 30, r = lerp(5, BR, q); kk.fillStyle = C.blue; kk.beginPath(); kk.arc(x, y, r, 0, 7); kk.fill(); }
    });
  } else {
    const q = P(u, cut, CYC);
    world(k, w, h, cam({ y: 1.2 }), {
      buildings: false, sun: null, trees: true,
      after: (kk) => { kk.fillStyle = C.blue; kk.beginPath(); kk.arc(BX, BY + q * 14, BR, 0, 7); kk.fill(); }
    });
  }
  // 形がそろっていることを示す点線の円
  const g = Math.max(0, 1 - Math.abs(u - cut) / 0.7);
  if (g > 0) { k.save(); k.translate(w / 2, h / 2); k.globalAlpha = g; k.strokeStyle = C.blue; k.lineWidth = 2; k.setLineDash([5, 4]); k.beginPath(); k.arc(BX, BY + (u > cut ? P(u, cut, CYC) * 14 : 0), BR + 9, 0, 7); k.stroke(); k.restore(); }
  return u < cut ? 'SHOT 1' : 'SHOT 2';
});
