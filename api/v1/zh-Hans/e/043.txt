// 甩镜头转场 / Whip pan
// 分类: camera (镜头运动与转场) · element 19 · 子类: 转场
// 示例效果: 相机快速横摇，在条状模糊最强时替换为下一场景。
// 效果规格:
//   - 衔接：1.7–2.5 s，历时 0.8 s；前 0.4 s 前镜头甩出，后 0.4 s 后镜头甩入并停稳。
//   - 前半段偏航 0→+0.45 rad，约 26°，二次缓入缓出；后半段偏航 -0.45 rad→0，缓动相同。
//   - 在进度 50% 处直接替换场景，此时模糊最强，掩盖切换。
//   - 模糊：对 30 fps 下跨越 ±1.6 帧的 9 次绘制取平均。
//   - 镜头高 1.4 m，除甩动外保持固定；A 中人物与 B 中汽车、城市均不动。
//   - 每 4.7 s 交替 A→B、B→A，往返 9.4 s；右上角显示 A → B / B → A。
// 适用场景: 适合保持动势地切换地点或话题。
// 标签: whip pan, swish pan, motion blur, fast pan, transition, energy
// 尺寸: 539x208 px · 周期: 9.4 s
// Needs: kit camera -> https://showheyohtaki.github.io/motion-prompt/api/v1/zh-Hans/kit/camera.js
// 动态示例: https://showheyohtaki.github.io/motion-prompt/#043
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D，调用前已清空并包裹 save/restore；w、h 为尺寸，u 为周期内秒数。画面仅由 u 决定。
// 以下是示例代码，CELLS、常量和辅助函数位于 kit 中。

// @element 19
// 19 ホイップパン（ブレで次の場面へ）
CELLS[18] = shot(abCell((k, w, h, u, A, B) => {
  const t0 = 1.7, t1 = 2.5;
  const at = (kk, s) => {   // s: 0..1 の進み。前半 A を右へ振り、後半 B が左から止まる
    if (s < .5) world(kk, w, h, cam({ y: 1.4, yaw: eIOs(s * 2) * 0.45 }), A === sceneA ? { person: { x: -0.6, z: 7 } } : SB);
    else world(kk, w, h, cam({ y: 1.4, yaw: -(1 - eIOs((s - .5) * 2)) * 0.45 }), B === sceneA ? { person: { x: -0.6, z: 7 } } : SB);
  };
  if (u < t0) return A(k, w, h);
  if (u > t1) return B(k, w, h);
  const s = P(u, t0, t1), ds = 1 / FPS / (t1 - t0);
  mblur(k, at, clamp(s - ds * 1.6), clamp(s + ds * 1.6), 9);
}));
