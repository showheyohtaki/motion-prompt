// 滚动进度条 / Scroll progress bar
// 分类: lp (落地页与网页效果) · element 07 · 子类: 滚动效果
// 示例效果: 逐段阅读时，顶部线从左侧增长，到页面底部时填满。
// 效果规格:
//   - 文章从 0→430 px 分四步滚动，每步 110 px、移动 0.45 s、停 0.4 s，使用三次缓入缓出。
//   - 3.75–4.55 s 从 430→0 返回。
//   - 视口顶部有高 6 px 的进度条，浅灰轨道、橙色填充。
//   - 填充宽度 = 511 px × scroll / 430，直接对应滚动比例，类似 CSS animation-timeline: scroll()。
// 适用场景: 适合长文章显示剩余阅读进度。
// 标签: progress-bar, reading-progress, scroll-progress, indicator, scroll-driven, article
// 尺寸: 539x208 px · 周期: 4.8 s
// Needs: kit lp -> https://showheyohtaki.github.io/motion-prompt/api/v1/zh-Hans/kit/lp.js
// 动态示例: https://showheyohtaki.github.io/motion-prompt/#079
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D，调用前已清空并包裹 save/restore；w、h 为尺寸，u 为周期内秒数。画面仅由 u 决定。
// 以下是示例代码，CELLS、常量和辅助函数位于 kit 中。

// @element 07
// 07 スクロールプログレスバー（読み進めた分だけ、画面上の線が伸びる）
CELLS[6] = (k, w, h, u) => {
  frame(k);
  const MAX = 430;
  const y = steps([[0.3, 0], [0.75, 110], [1.15, 110], [1.6, 220], [2.0, 220], [2.45, 330], [2.85, 330], [3.3, MAX], [3.75, MAX], [4.55, 0]], u);
  view(k, () => {
    k.save(); k.translate(0, -y);
    txt(k, 'はじめてのWebデザイン', 70, 44, `800 20px ${JP}`, C.ink); bars(k, 70, 56, 120, 1, C.faint, 5);
    pic(k, 70, 74, 371, 112, 0, 8);
    for (let b = 0; b < 4; b++) { const y0 = 204 + b * 82; k.fillStyle = C.ink; rr(k, 70, y0, 140, 9, 4.5); k.fill(); bars(k, 70, y0 + 22, 371, 4, C.faint, 6, 12); }
    k.restore();
    k.fillStyle = C.faint; k.fillRect(0, 0, VW, 6);
    k.fillStyle = C.acc; k.fillRect(0, 0, VW * (y / MAX), 6);
    sbar(k, y / MAX, 0.3);
  });
};
