// 无限滚动 / Infinite scroll
// 分类: lp (落地页与网页效果) · element 19 · 子类: 加载、导航与循环动效
// 示例效果: 滚到列表末尾出现加载圆环，然后追加三个条目以继续阅读。
// 效果规格:
//   - 列表项 331 × 44、圆角半径 8、间距 52；末项下方是半径 10、每 0.8 s 转一圈的加载圆环。
//   - 滚动：0.25–0.85 s 为 38→116 px，1.15–1.75 s 为 116→194 px，三次缓入缓出；停在列表末尾可见的位置。
//   - 2.6 s 追加三项，各自在 0.35 s 内按三次缓出从下方 10 px 升起，不透明度 0→1；加载圆环移至新的末尾。
//   - 每周期前进三项，即 156 px；内容每三项重复，因此周期末尾与开头一致。
// 适用场景: 适合无需分页、持续阅读的信息流和新闻列表。
// 标签: infinite-scroll, endless-scroll, load-more, feed, pagination, spinner, list
// 尺寸: 539x208 px · 周期: 4.8 s
// Needs: kit lp -> https://showheyohtaki.github.io/motion-prompt/api/v1/zh-Hans/kit/lp.js
// 动态示例: https://showheyohtaki.github.io/motion-prompt/#091
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D，调用前已清空并包裹 save/restore；w、h 为尺寸，u 为周期内秒数。画面仅由 u 决定。
// 以下是示例代码，CELLS、常量和辅助函数位于 kit 中。

// @element 19
// 19 無限スクロール（いちばん下まで来ると読み込み表示が出て、続きが足される）
CELLS[18] = (k, w, h, u) => {
  frame(k);
  // 1周期で 3件（156px）ぶん進む。周期の終わりは始まりと同じ見た目（内容は3件ごとにくり返す）
  const PITCH = 52, BATCH = 3, L0 = 6, TL = 2.6;
  const y = steps([[0.25, 38], [0.85, 116], [1.15, 116], [1.75, 194], [4.8, 194]], u);
  const n = u < TL ? L0 : L0 + BATCH, end = 10 + n * PITCH;
  view(k, () => {
    k.fillStyle = C.soft; k.fillRect(0, 0, VW, VH);
    k.save(); k.translate(0, -y);
    for (let i = 0; i < n; i++) {
      const top = 10 + i * PITCH, v = i % 3, fresh = i >= L0 ? eOut(P(u, TL, TL + 0.35)) : 1;
      if (top - y > VH || top + 44 - y < 0) continue;
      k.save(); k.globalAlpha = fresh; k.translate(0, 10 * (1 - fresh));
      k.fillStyle = '#fff'; rr(k, 90, top, 331, 44, 8); k.fill(); pic(k, 98, top + 6, 48, 32, v, 5);
      bars(k, 158, top + 12, [180, 140, 200][v], 1, C.ink, 7); bars(k, 158, top + 26, [120, 150, 100][v], 1, C.faint, 6);
      k.restore();
    }
    spinner(k, VW / 2, end + 18, 10, u);
    k.restore();
  });
};
