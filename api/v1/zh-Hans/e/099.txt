// 剪影 / Silhouette
// 分类: photo (摄影效果、光线与风格) · element 03 · 子类: 布光与时段
// 示例效果: 按明亮晚霞曝光，人物、建筑、路灯和树变为无细节的黑色形状，仅天空保持明亮。
// 效果规格:
//   - 场景 1：人物半身像位于 y 150 的地平线前，头部中心 x 270、y 78，半径 29 px，无五官或头发。
//   - 天空从 #FAF9F7 变为顶部 #FFEDC4，地平线附近 y 92–150 为 #FFCF73 色带。
//   - 天空以外变为 #2A2D36，远山 #4A4546，人物 #22252D。
//   - 太阳半径 20 px、x 404，从 y 60 下沉到 y 128，一半被山挡住。
//   - 右上曝光补偿 ±0.0 EV→−2.0 EV。
//   - 时间：0.4–1.4 s 效果渐入，保持到 4.0 s，至 4.8 s 消退；每轮 5 s，三次缓入缓出，约 cubic-bezier(0.65,0,0.35,1)。
// 适用场景: 适合夕阳或明窗前，依靠轮廓和姿态表达的画面。
// 标签: silhouette, backlit, sunset, contre-jour, black shape, underexposed
// 尺寸: 539x208 px · 周期: 5 s
// Needs: kit photo -> https://showheyohtaki.github.io/motion-prompt/api/v1/zh-Hans/kit/photo.js
// 动态示例: https://showheyohtaki.github.io/motion-prompt/#099
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D，调用前已清空并包裹 save/restore；w、h 为尺寸，u 为周期内秒数。画面仅由 u 决定。
// 以下是示例代码，CELLS、常量和辅助函数位于 kit 中。

// @element 03
// シルエット: 露出を明るい空に合わせると、人物も建物も木も黒い形だけになる
CELLS[2] = cell((k, u) => {
  const a = env(u), INK = '#2A2D36';
  const SILH = mapPal(BASE, () => INK);
  Object.assign(SILH, { sky: '#FFEDC4', sky2: '#FFCF73', far: '#4A4546', sun: BASE.sun, glow: BASE.glow });
  stagePortrait(k, pal(BASE, SILH, a), { sun: [404, lerp(60, 128, a), 20, a] });
  bust(k, PX, PY, PR, mixc(BASE.person, '#22252D', a));
  hud(k, `${(-2 * a).toFixed(1).replace('-0.0', '±0.0')} EV`);
});
