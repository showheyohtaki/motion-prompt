// 蝴蝶光 / Butterfly lighting
// 分类: photo (摄影效果、光线与风格) · element 06 · 子类: 布光与时段
// 示例效果: 正前方高处灯亮起，脸部两侧均匀受光，下巴下仅留一小块对称阴影。
// 效果规格:
//   - 场景 1：人物半身像位于 y 150 的地平线前，头部中心 x 270、y 78，半径 29 px，无五官或头发。 背景变暗，天空 #E7E5E1。
//   - 头部完全受光，#FFC75A，没有阴影侧。
//   - 下巴下阴影为椭圆，中心 y 78+1.06R，宽 0.72R、高 0.38R，R = 29，约 21×11 px，颜色 #33405A。
//   - 身体中心 ±0.8×56 px 内受光，外侧细条保留底色；示意图灯在相机侧 0°，右上显示 0° · ↑45°。
//   - 时间：0.4–1.4 s 效果渐入，保持到 4.0 s，至 4.8 s 消退；每轮 5 s，三次缓入缓出，约 cubic-bezier(0.65,0,0.35,1)。
// 适用场景: 适合突出面容的美妆和时尚人像。
// 标签: butterfly lighting, paramount lighting, beauty lighting, portrait lighting, overhead light, glamour
// 尺寸: 539x208 px · 周期: 5 s
// Needs: kit photo -> https://showheyohtaki.github.io/motion-prompt/api/v1/zh-Hans/kit/photo.js
// 动态示例: https://showheyohtaki.github.io/motion-prompt/#102
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D，调用前已清空并包裹 save/restore；w、h 为尺寸，u 为周期内秒数。画面仅由 u 决定。
// 以下是示例代码，CELLS、常量和辅助函数位于 kit 中。

// @element 06
// バタフライライティング: 顔の正面・高い位置からの光。顔は左右対称に明るく、あごの下にだけ小さな影
CELLS[5] = cell((k, u) => {
  const a = env(u), R = PR, W = torsoW(R);
  stagePortrait(k, pal(BASE, DIM, a));
  const lit = mixc(BASE.person, BASE.lit, a), sh = mixc(BASE.person, BASE.shade, a);
  bust(k, PX, PY, R, BASE.person, kk => {
    torso(kk, PX, PY, R, PX - 0.8 * W, PX + 0.8 * W, lit);
    kk.fillStyle = sh; kk.beginPath(); kk.ellipse(PX, PY + R * 1.06, R * 0.72, R * 0.38, 0, 0, TAU); kk.fill();    // あごの下の影（左右対称）
    kk.fillStyle = lit; kk.beginPath(); kk.arc(PX, PY, R, 0, TAU); kk.fill();
  });
  plan(k, [[0, a]]);
  hud(k, '0° · ↑45°');
});
