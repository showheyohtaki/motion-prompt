// 追随拍摄 / Panning shot
// 分类: photo (摄影效果、光线与风格) · element 12 · 子类: 镜头与相机效果
// 示例效果: 相机跟随汽车，使车保持清晰居中，只有背后的建筑、树和路灯横向拖影。
// 效果规格:
//   - 街角建筑、树和路灯位于 y 132–162 道路后，无前景大人物。
//   - 无效果时汽车在 5 s 内从 x −80 行至 620；有效果时保持 x 270 ±10 px。
//   - 背景在横向 46 px 范围分布绘制 12 次再取平均，展开范围 0→46 px。
//   - 车轮辐条每 5 s 旋转 12 圈。
//   - 右上快门速度 1/1000→1/15。
//   - 时间：0.4–1.4 s 效果渐入，保持到 4.0 s，至 4.8 s 消退；每轮 5 s，三次缓入缓出，约 cubic-bezier(0.65,0,0.35,1)。
// 适用场景: 适合静态照片中表现汽车、自行车或跑者的速度。
// 标签: panning, panning shot, motion blur, speed, sports photography, slow shutter
// 尺寸: 539x208 px · 周期: 5 s
// Needs: kit photo -> https://showheyohtaki.github.io/motion-prompt/api/v1/zh-Hans/kit/photo.js
// 动态示例: https://showheyohtaki.github.io/motion-prompt/#108
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D，调用前已清空并包裹 save/restore；w、h 为尺寸，u 为周期内秒数。画面仅由 u 决定。
// 以下是示例代码，CELLS、常量和辅助函数位于 kit 中。

// @element 12
// 流し撮り: 走る車を追ってカメラを振ると、車はくっきり、背景だけ横に流れる
CELLS[11] = cell((k, u) => {
  const a = env(u);
  const drive = lerp(-80, 620, u / CYC), x = lerp(drive, 270 + 10 * Math.sin(u / CYC * TAU), a);
  const S = 46 * a, n = S < 0.5 ? 1 : 12;
  for (let i = 0; i < n; i++) {   // 背景を横にずらして重ね、平均をとる（流れ）
    k.save(); k.globalAlpha = 1 / (i + 1); k.translate(n > 1 ? lerp(-S / 2, S / 2, i / (n - 1)) : 0, 0); stageStreet(k, BASE, { car: false, person: false }); k.restore();
  }
  const D = pen(k); car(D, x, BASE, { head: BASE.lit });
  // 車輪の回転（1周に12回転＝継ぎ目なし）
  k.save(); k.strokeStyle = BASE.tire; k.lineWidth = 1.6;
  for (const wx of [x - 21, x + 21]) { const t = u / CYC * TAU * 12; k.beginPath(); k.moveTo(wx - Math.cos(t) * 5, 156 - Math.sin(t) * 5); k.lineTo(wx + Math.cos(t) * 5, 156 + Math.sin(t) * 5); k.stroke(); }
  k.restore();
  hud(k, a > 0.5 ? '1/15' : '1/1000');
});
